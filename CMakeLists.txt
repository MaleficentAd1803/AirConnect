cmake_minimum_required(VERSION 3.21)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/src/version.h" VERSION_LINE REGEX "^#define AC_VERSION ")
string(REGEX MATCH "[0-9]+\.[0-9]+\.[0-9]+" AIRCONNECT_VERSION "${VERSION_LINE}")
project(AirConnect VERSION ${AIRCONNECT_VERSION} LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 HAVE_LIBPLIST=1 HAVE_PLAYFAIR=1 LIBPLIST_STATIC=1)
set(UP ${CMAKE_CURRENT_SOURCE_DIR}/upstream/AirPlay-Windows)
set(DEPS ${CMAKE_CURRENT_SOURCE_DIR}/dependencies)
set(RUNTIME_DIR "${DEPS}/runtime" CACHE PATH "Native runtime DLLs")
file(GLOB PLIST_C CONFIGURE_DEPENDS ${DEPS}/libplist/src/*.c ${DEPS}/libplist/libcnary/*.c)
list(FILTER PLIST_C EXCLUDE REGEX "cnary.c$")
add_library(plist STATIC ${PLIST_C})
target_include_directories(plist PRIVATE ${DEPS}/libplist/src ${DEPS}/libplist/libcnary/include PUBLIC ${DEPS}/libplist/include)
target_compile_definitions(plist PRIVATE LIBPLIST_STATIC PACKAGE_VERSION="2.4.0")
file(GLOB AIRPLAY_CXX CONFIGURE_DEPENDS ${UP}/src/airplay/*.cpp ${UP}/src/net/*.cpp ${UP}/src/crypto/*.cpp ${UP}/src/audio/*.cpp)
list(FILTER AIRPLAY_CXX EXCLUDE REGEX "/(audio_output|mirror_listener|identity|rtsp_parser|tcp_server|server)\\.cpp$")
file(GLOB PLAYFAIR_C ${UP}/third_party/playfair/*.c)
add_library(receiver STATIC ${AIRPLAY_CXX} ${PLAYFAIR_C}
 src/core/mdns_service.cpp src/core/mirror_listener.cpp src/core/identity.cpp src/core/rtsp_parser.cpp src/core/tcp_server.cpp src/core/server.cpp ${UP}/src/video/h264_decoder.cpp
 ${UP}/src/video/hls_player.cpp src/video/video_recorder.cpp
 ${UP}/src/video/gpu_recording.cpp ${UP}/third_party/fairplay_blobs_real.cpp
 src/platform/audio_output.cpp src/platform/capture.cpp src/platform/window_state.cpp src/platform/network_policy.cpp src/video/video_renderer.cpp src/core/log.cpp src/core/preferences.cpp src/core/updates.cpp)
target_include_directories(receiver PUBLIC src ${UP}/src ${DEPS}/libplist/include ${DEPS}/include PRIVATE ${UP}/third_party ${UP}/third_party/playfair)
target_compile_definitions(receiver PUBLIC DNS_REQUEST_PENDING=9506)
target_link_libraries(receiver PUBLIC plist
 "${DEPS}/lib/libcrypto-3-x64.a"
 "${DEPS}/lib/avcodec-61.a" "${DEPS}/lib/avformat-61.a"
 "${DEPS}/lib/avutil-59.a" "${DEPS}/lib/swscale-8.a"
 ws2_32 iphlpapi dnsapi "${DEPS}/lib/dnsservice.a" d3d11 dxgi d2d1 dwrite dwmapi
 ole32 oleaut32 shell32 uuid winmm propsys comctl32 shlwapi windowscodecs crypt32 advapi32 winhttp bcrypt)
if(MINGW)
 target_compile_options(receiver PUBLIC -fno-char8_t "$<$<COMPILE_LANGUAGE:CXX>:-include${CMAKE_CURRENT_SOURCE_DIR}/src/platform/compat.h>" PRIVATE -Wall -Wextra -Wno-unknown-pragmas)
 target_link_options(receiver PUBLIC -static-libgcc -static-libstdc++)
endif()
add_executable(AirConnect WIN32 src/app.cpp src/core/receiver.cpp src/ui/home.cpp src/app.rc)
target_link_libraries(AirConnect PRIVATE receiver)
if(MINGW)
 target_link_options(AirConnect PRIVATE -municode)
endif()
enable_testing()
add_executable(core-tests tests/core_tests.cpp)
target_link_libraries(core-tests PRIVATE receiver)
add_test(NAME native-core COMMAND core-tests)
add_executable(protocol-tests ${UP}/tools/test_shutdown.cpp)
target_link_libraries(protocol-tests PRIVATE receiver)
add_test(NAME transport-lifecycle COMMAND protocol-tests)
add_executable(audio-timing-tests tests/audio_timing.cpp)
target_link_libraries(audio-timing-tests PRIVATE receiver)
add_test(NAME audio-clock COMMAND audio-timing-tests)
add_executable(media-tests tests/media_tests.cpp)
target_link_libraries(media-tests PRIVATE receiver)
add_test(NAME native-media COMMAND media-tests)
add_executable(update-tests tests/update_tests.cpp)
target_link_libraries(update-tests PRIVATE receiver)
add_test(NAME update-policy COMMAND update-tests)
add_custom_target(copy-runtime ALL COMMAND ${CMAKE_COMMAND} -E copy_directory "${RUNTIME_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
foreach(t AirConnect core-tests protocol-tests audio-timing-tests media-tests update-tests)
 add_dependencies(${t} copy-runtime)
endforeach()
install(TARGETS AirConnect RUNTIME DESTINATION .)
install(DIRECTORY ${RUNTIME_DIR}/ DESTINATION . FILES_MATCHING PATTERN "*.dll")
install(FILES LICENSE THIRD_PARTY_NOTICES.md README.md DESTINATION .)
